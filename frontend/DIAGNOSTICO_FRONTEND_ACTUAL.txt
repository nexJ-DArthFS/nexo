function formatDate(value) {
  if (!value) return "Sin fecha";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "Fecha no vÃ¡lida";

  return new Intl.DateTimeFormat("es-EC", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

function formatScope(scope) {
  if (!scope) return "No especificado";

  if (typeof scope === "string") {
    return scope.replaceAll("_", " ");
  }

  if (typeof scope === "object") {
    const type = scope.type || scope.level || "personal";

    const target =
      scope.module ||
      (scope.resourceType && scope.resourceId
        ? `${scope.resourceType}/${scope.resourceId}`
        : scope.resource || scope.id);

    return target ? `${type}: ${target}` : type;
  }

  return String(scope);
}

function getAssignmentStatus(assignment) {
  if (assignment.effective) {
    return {
      label: "Efectiva",
      className: "effective",
    };
  }

  if (assignment.expired) {
    return {
      label: "Vencida",
      className: "expired",
    };
  }

  if (!assignment.active) {
    return {
      label: "Inactiva",
      className: "inactive",
    };
  }

  if (!assignment.scopeAllowed) {
    return {
      label: "Alcance invÃ¡lido",
      className: "invalid",
    };
  }

  if (!assignment.role) {
    return {
      label: "Rol no disponible",
      className: "invalid",
    };
  }

  if (assignment.role.active === false) {
    return {
      label: "Rol inactivo",
      className: "inactive",
    };
  }

  return {
    label: "No efectiva",
    className: "inactive",
  };
}

// Alcances permitidos por rol.
// El backend conserva la validaciÃ³n definitiva.
function getScopeTypeForRole(roleKey) {
  const scopes = {
    super_admin: "global",
    admin: "global",
    module_manager: "module",
    moderator: "resource",
    user: "global",
  };

  return scopes[roleKey] || "";
}

function createEmptyAssignmentForm() {
  return {
    roleId: "",
    module: "",
    resourceType: "",
    resourceId: "",
    expiresAt: "",
  };
}

function buildAssignmentScope(role, form) {
  const scopeType = getScopeTypeForRole(role?.key);

  if (scopeType === "global") {
    return { type: "global" };
  }

  if (scopeType === "module") {
    return {
      type: "module",
      module: form.module.trim(),
    };
  }

  if (scopeType === "resource") {
    return {
      type: "resource",
      resourceType: form.resourceType.trim(),
      resourceId: form.resourceId.trim(),
    };
  }

  throw new Error(
    "No se reconoce el alcance permitido para este rol."
  );
}

function toDateTimeLocal(value) {
  if (!value) return "";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "";

  const offset = date.getTimezoneOffset() * 60000;

  return new Date(date.getTime() - offset)
    .toISOString()
    .slice(0, 16);
}

function formatDate(value) {

    return "Fecha no disponible";

  return new Intl.DateTimeFormat("es-EC", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(date);
}

async function request(endpoint, options = {}) {
  const response = await fetch(`${API_URL}${endpoint}`, {
    credentials: "include",
    cache: "no-store",
    ...options,
    headers: {
      Accept: "application/json",
      ...(options.body ? { "Content-Type": "application/json" } : {}),
      ...options.headers,
    },
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      data.message ||
      data.error ||
      `Error en la peticiÃ³n (${response.status}).`
    );
  }

  return data;
}
