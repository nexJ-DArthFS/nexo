const ROLE_ALLOWED_SCOPES = Object.freeze({
  super_admin: ['global'],
  admin: ['global'],
  module_manager: ['module'],
  moderator: ['resource'],
  user: ['global']
});

// ======================================================
// VALIDACIÃ“N DE IDENTIFICADORES
// ======================================================

function toObjectId(value) {
  if (value instanceof ObjectId) {
    return value;
  }

  if (
    typeof value !== 'string' ||
    !ObjectId.isValid(value) ||
    value.length !== 24
  ) {
    throw new Error('Identificador de usuario invÃ¡lido.');
  }

  return new ObjectId(value);
}

function isAssignmentActive(assignment, now = new Date()) {
  if (!assignment || assignment.active !== true) {
    return false;
  }

  if (
    assignment.expiresAt === null ||
    assignment.expiresAt === undefined
  ) {
    return true;
  }

  const expiration = new Date(assignment.expiresAt);

  if (Number.isNaN(expiration.getTime())) {
    return false;
  }

  return expiration > now;
}

// ======================================================
// VALIDACIÃ“N DE ESTRUCTURA DE ALCANCES
// ======================================================

function isValidScope(scope) {
  if (
    !scope ||
    typeof scope !== 'object' ||
    Array.isArray(scope)
  ) {
    return false;
  }

  if (scope.type === 'global') {
    return true;
  }

  if (scope.type === 'module') {
    return (
      typeof scope.module === 'string' &&
      scope.module.trim().length > 0
    );
  }

  if (scope.type === 'resource') {
    return (
      typeof scope.resourceType === 'string' &&
      scope.resourceType.trim().length > 0 &&
      scope.resourceId !== undefined &&
      scope.resourceId !== null &&
      String(scope.resourceId).trim().length > 0
    );
  }

  return false;
}

// ======================================================
// VALIDACIÃ“N DE ALCANCES PERMITIDOS POR ROL
// ======================================================

function isRoleScopeAllowed(roleKey, scope) {
  if (!isValidScope(scope)) {
    return false;
  }

  const allowedScopes = ROLE_ALLOWED_SCOPES[roleKey];

  if (!allowedScopes) {
    return false;
  }

  return allowedScopes.includes(scope.type);
}

// ======================================================
// VALIDACIÃ“N DE DESTINOS
// ======================================================

function isValidTarget(target) {
  return (
    target !== null &&
    typeof target === 'object' &&
    !Array.isArray(target)
  );
}

// ======================================================
// COMPROBACIÃ“N DE ALCANCES
// ======================================================

function scopeAllows(scope, target = {}) {
  if (
    !isValidScope(scope) ||
    !isValidTarget(target)
  ) {
    return false;
  }

  // Alcance global.
  if (scope.type === 'global') {
    return true;
  }

  // Alcance por mÃ³dulo.
  if (scope.type === 'module') {
    return (
      typeof target.module === 'string' &&
      target.module.trim().length > 0 &&
      scope.module === target.module
    );
  }

  // Alcance por recurso.
  if (scope.type === 'resource') {
    if (
      typeof target.resourceType !== 'string' ||
      !target.resourceType.trim() ||
      target.resourceId === undefined ||
      target.resourceId === null
    ) {
      return false;
    }

    return (
      scope.resourceType === target.resourceType &&
      String(scope.resourceId) === String(target.resourceId)
    );
  }

  return false;
}

async function getUserAssignments(db, userId) {
  const id = toObjectId(userId);

  return assignments.filter(assignment =>

  );
}

  }

  const roleMap = new Map(
    roles.map(role => [role._id.toString(), role])
  );

  // Conservar solamente asignaciones con alcances
  // permitidos para sus respectivos roles.

  const validAssignments = assignments.filter(assignment => {
    if (!(assignment.roleId instanceof ObjectId)) {
      return false;
    }

    const role = roleMap.get(assignment.roleId.toString());

    if (!role) {
      return false;
    }

    return isRoleScopeAllowed(role.key, assignment.scope);
  });

  if (validAssignments.length === 0) {
    return [];
  }

  const activeRoleIds = [
    ...new Map(
      validAssignments.map(assignment => [
        assignment.roleId.toString(),
        assignment.roleId
      ])
    ).values()
  ];

  // Consultar relaciones entre roles y permisos.

  const relations = await db.collection('role_permissions')
    .find({
      roleId: { $in: activeRoleIds }
    })
    .toArray();

  const validRelations = relations.filter(relation =>
    relation.roleId instanceof ObjectId &&
    relation.permissionId instanceof ObjectId
  );

  if (validRelations.length === 0) {
    return [];
  }

  // Obtener identificadores Ãºnicos de permisos.

  const permissionIds = [
    ...new Map(
      validRelations.map(relation => [
        relation.permissionId.toString(),
        relation.permissionId
      ])
    ).values()
  ];

  // Consultar permisos activos.

  const permissions = await db.collection('permissions')
    .find({
      _id: { $in: permissionIds },
      active: { $ne: false }
    })
    .toArray();

  const permissionMap = new Map(
    permissions.map(permission => [
      permission._id.toString(),
      permission
    ])
  );

  const effective = [];

  // Combinar permisos conservando cada asignaciÃ³n
  // y su alcance independiente.

  for (const assignment of validAssignments) {
    const role = roleMap.get(assignment.roleId.toString());

    if (!role) {
      continue;
    }

    for (const relation of validRelations) {
      if (
        relation.roleId.toString() !== role._id.toString()
      ) {
        continue;
      }

      const permission = permissionMap.get(
        relation.permissionId.toString()
      );

      if (
        !permission ||
        typeof permission.key !== 'string' ||
        typeof permission.module !== 'string'
      ) {
        continue;
      }

      effective.push({
        key: permission.key,
        module: permission.module,
        roleKey: role.key,
        scope: assignment.scope,
        assignmentId: assignment._id
      });
    }
  }

  return effective;
}

async function hasPermission(
  db,
  userId,
  permissionKey,
  target = {}
) {
  if (
    typeof permissionKey !== 'string' ||
    !permissionKey.trim() ||
    !isValidTarget(target)
  ) {
    return false;
  }

  if (
    typeof target.module !== 'string' ||
    !target.module.trim()
  ) {
    return false;
  }

  const id = toObjectId(userId);

  // Comprobar estado actual del usuario.

  const user = await db.collection('users').findOne(
    { _id: id },
    {
      projection: {
        status: 1
      }
    }
  );

  if (!user || user.status !== 'active') {
    return false;
  }

  const permissions = await getEffectivePermissions(db, id);

  return permissions.some(permission => {
    // Coincidencia de permiso.

    if (permission.key !== permissionKey) {
      return false;
    }

    // Coincidencia de mÃ³dulo.

    if (permission.module !== target.module) {
      return false;
    }

    // Verificar el alcance individual.

    return scopeAllows(permission.scope, target);
  });
}

function validateCatalog() {
  if (!Array.isArray(ROLES) || !Array.isArray(PERMISSIONS)) {
    throw new Error('ROLES y PERMISSIONS deben ser arrays.');
  }

  if (
    !ROLE_PERMISSIONS ||
    typeof ROLE_PERMISSIONS !== 'object' ||
    Array.isArray(ROLE_PERMISSIONS)
  ) {
    throw new Error('ROLE_PERMISSIONS debe ser un objeto vÃ¡lido.');
  }

  const roleKeys = new Set(ROLES.map(role => role.key));
  const permissionKeys = new Set(
    PERMISSIONS.map(permission => permission.key)
  );

  if (roleKeys.size !== ROLES.length) {
    throw new Error('El catÃ¡logo contiene roles duplicados.');
  }

  if (permissionKeys.size !== PERMISSIONS.length) {
    throw new Error('El catÃ¡logo contiene permisos duplicados.');
  }

  for (const role of ROLES) {
    if (
      !role ||
      typeof role.key !== 'string' ||
      !role.key.trim() ||
      typeof role.name !== 'string' ||
      !role.name.trim()
    ) {
      throw new Error('Todos los roles deben tener key y name vÃ¡lidos.');
    }

    if (!Object.prototype.hasOwnProperty.call(ROLE_PERMISSIONS, role.key)) {
      throw new Error(
        `El rol ${role.key} no tiene asignaciones definidas.`
      );
    }
  }

  for (const permission of PERMISSIONS) {
    if (
      !permission ||
      typeof permission.key !== 'string' ||
      !permission.key.trim() ||
      typeof permission.name !== 'string' ||
      !permission.name.trim() ||
      typeof permission.module !== 'string' ||
      !permission.module.trim()
    ) {
      throw new Error(
        'Todos los permisos deben tener key, name y module vÃ¡lidos.'
      );
    }
  }

  for (const roleKey of Object.keys(ROLE_PERMISSIONS)) {
    if (!roleKeys.has(roleKey)) {
      throw new Error(`Rol no definido en ROLES: ${roleKey}`);
    }

    const assignedPermissions = ROLE_PERMISSIONS[roleKey];

    if (!Array.isArray(assignedPermissions)) {
      throw new Error(
        `Los permisos del rol ${roleKey} deben ser un array.`
      );
    }

    if (new Set(assignedPermissions).size !== assignedPermissions.length) {
      throw new Error(
        `El rol ${roleKey} contiene permisos duplicados.`
      );
    }

    for (const permissionKey of assignedPermissions) {
      if (!permissionKeys.has(permissionKey)) {
        throw new Error(
          `Permiso no definido en PERMISSIONS: ${permissionKey}`
        );
      }
    }
  }

async function synchronizeCatalog(db, session) {
  const rolesCollection = db.collection('roles');
  const permissionsCollection = db.collection('permissions');
  const relationsCollection = db.collection('role_permissions');

  const options = session ? { session } : {};
  const now = new Date();

  // ====================================================
  // 1. CREAR O ACTUALIZAR ROLES
  // ====================================================

  for (const role of ROLES) {
    await rolesCollection.updateOne(
      { key: role.key },
      {
        $set: {
          name: role.name,
          description: role.description,
          protected: role.protected === true,
          active: role.active !== false,
          updatedAt: now
        },
        $setOnInsert: {
          createdAt: now
        }
      },
      {
        ...options,
        upsert: true
      }
    );
  }

  // ====================================================
  // 2. CREAR O ACTUALIZAR PERMISOS
  // ====================================================

  for (const permission of PERMISSIONS) {
    await permissionsCollection.updateOne(
      { key: permission.key },
      {
        $set: {
          name: permission.name,
          module: permission.module,
          description: permission.description,
          active: permission.active !== false,
          updatedAt: now
        },
        $setOnInsert: {
          createdAt: now
        }
      },
      {
        ...options,
        upsert: true
      }
    );
  }

  // ====================================================
  // 3. RECUPERAR ROLES Y PERMISOS
  // ====================================================

  const roles = await rolesCollection.find(
    { key: { $in: ROLES.map(role => role.key) } },
    options
  ).toArray();

  const permissions = await permissionsCollection.find(
    { key: { $in: PERMISSIONS.map(permission => permission.key) } },
    options
  ).toArray();

  const roleMap = new Map(
    roles.map(role => [role.key, role])
  );

  const permissionMap = new Map(
    permissions.map(permission => [permission.key, permission])
  );

  if (roleMap.size !== ROLES.length) {
    throw new Error(
      'No se pudieron recuperar todos los roles del catÃ¡logo.'
    );
  }

  if (permissionMap.size !== PERMISSIONS.length) {
    throw new Error(
      'No se pudieron recuperar todos los permisos del catÃ¡logo.'
    );
  }

  // ====================================================
  // 4. SINCRONIZAR RELACIONES
  // ====================================================

  let relationsCreated = 0;
  let relationsDeleted = 0;

  const expectedRoleIds = roles.map(role => role._id);

  for (const [roleKey, permissionKeys] of Object.entries(
    ROLE_PERMISSIONS
  )) {
    const role = roleMap.get(roleKey);

    if (!role) {
      throw new Error(`No se encontrÃ³ el rol: ${roleKey}`);
    }

    const expectedPermissionIds = [];

    for (const permissionKey of permissionKeys) {
      const permission = permissionMap.get(permissionKey);

      if (!permission) {
        throw new Error(
          `No se encontrÃ³ el permiso: ${permissionKey}`
        );
      }

      expectedPermissionIds.push(permission._id);

      const result = await relationsCollection.updateOne(
        {
          roleId: role._id,
          permissionId: permission._id
        },
        {
          $setOnInsert: {
            roleId: role._id,
            permissionId: permission._id,
            createdAt: now
          }
        },
        {
          ...options,
          upsert: true
        }
      );

      if (result.upsertedCount === 1) {
        relationsCreated++;
      }
    }

    // Eliminar relaciones que no estÃ©n en el catÃ¡logo.

    const deleteResult = await relationsCollection.deleteMany(
      {
        roleId: role._id,
        permissionId: {
          $nin: expectedPermissionIds
        }
      },
      options
    );

    relationsDeleted += deleteResult.deletedCount;
  }

  // Eliminar relaciones de roles que ya no existen
  // en el catÃ¡logo autoritativo.

  const staleRolesResult = await relationsCollection.deleteMany(
    {
      roleId: {
        $nin: expectedRoleIds
      }
    },
    options
  );

  relationsDeleted += staleRolesResult.deletedCount;

  // ====================================================
  // 5. RESULTADO
  // ====================================================

  return {
    roles: ROLES.length,
    permissions: PERMISSIONS.length,
    relationsCreated,
    relationsDeleted,
    rolesSynchronized: Object.keys(ROLE_PERMISSIONS).length
  };
}

const ALLOWED_STATUSES = [
  'pending',
  'active',
  'suspended',
  'disabled'
];

function createUsersService(db) {
  const repository = createUsersRepository(db);
  const audit = createAuditService(db);

  function validateObjectId(id, fieldName = 'ID') {
    if (!id || !ObjectId.isValid(id)) {
      const error = new Error(`${fieldName} invÃ¡lido.`);
      error.status = 400;
      throw error;
    }

    return new ObjectId(id);
  }

  async function withTransaction(callback) {
    const session = db.client.startSession();

    try {
      let result;

      await session.withTransaction(async () => {
        result = await callback(session);
      });

      return result;
    } finally {
      await session.endSession();
    }
  }

  async function getById(id) {
    const validId = validateObjectId(id);

    const user = await repository.findById(validId);

    if (!user) {
      const error = new Error('Usuario no encontrado.');
      error.status = 404;
      throw error;
    }

    return user;
  }

  async function list(options = {}) {
    const skip = Math.max(0, Number(options.skip) || 0);
    const limit = Math.min(
      100,
      Math.max(1, Number(options.limit) || 50)
    );

    const status = options.status;

    if (status && !ALLOWED_STATUSES.includes(status)) {
      const error = new Error('Estado de usuario invÃ¡lido.');
      error.status = 400;
      throw error;
    }

    const filter = status ? { status } : {};

    const [users, total] = await Promise.all([
      repository.list({ skip, limit, status }),
      repository.count(filter)
    ]);

    return {
      users,
      pagination: {
        total,
        skip,
        limit
      }
    };
  }

  async function updateProfile(id, changes, actorId) {
    const userId = validateObjectId(id, 'ID de usuario');
    const actorObjectId = validateObjectId(actorId, 'ID del actor');

    const allowedChanges = {};

    if (typeof changes?.name === 'string') {
      const name = changes.name.trim();

      if (name.length < 2 || name.length > 80) {
        const error = new Error(

        error.status = 400;
        throw error;
      }

      allowedChanges.name = name;
    }

    if (Object.keys(allowedChanges).length === 0) {
      const error = new Error('No hay cambios vÃ¡lidos para aplicar.');
      error.status = 400;
      throw error;
    }

    return withTransaction(async (session) => {
      const users = db.collection('users');

      const user = await users.findOne(

          projection: { passwordHash: 0 },

        error.status = 404;
        throw error;
      }

      await users.updateOne(
        { _id: userId },
        {
          $set: {
            ...allowedChanges,
            updatedAt: new Date()
          }
        },
        { session }
      );

      await audit.record({
        actorId: actorObjectId,
        action: 'users.profile.updated',

          type: 'user',
          id: userId
        },

  async function updateStatus(id, status, actorId, req = {}) {
    const userId = validateObjectId(id, 'ID de usuario');
    const actorObjectId = validateObjectId(actorId, 'ID del actor');

    if (!ALLOWED_STATUSES.includes(status)) {

              },
              {
                $count: 'total'

function classifyFragment(text) {
  for (const rule of rules) {
    if (rule.patterns.some(pattern => pattern.test(text))) {
      return {
        category: rule.category,
        label: rule.label
      };
    }
  }

  return {
    category: 'review',
    label: 'Por revisar'
  };
}

function splitIntoFragments(text) {
  return text
    .replace(/\r\n/g, '\n')
    .split(/\n+|(?<=[.!?])\s+/)
    .map(fragment => fragment.trim())
    .filter(Boolean);
}

    decisions: [],

    knowledge: [],
    review: []

    totalFragments: items.length,
